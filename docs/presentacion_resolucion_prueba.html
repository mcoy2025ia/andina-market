-- ============================================================================
-- Parte 1, punto 1 - T-SQL (SQL Server / Fabric Warehouse)
-- Esquema: silver.pedidos, silver.clientes, silver.v_cliente_maestro (ver ddl_tsql.sql)
-- Definiciones:
--   venta neta    = valor_bruto - descuento, solo estado ENTREGADO, por fecha_pedido
--   persona       = id_cliente_maestro (une los clientes duplicados por documento)
--   cohorte       = mes de registro más antiguo de la persona
--   primer pedido = primer pedido ENTREGADO de la persona (por fecha)
--   recompra N    = otro pedido ENTREGADO en fecha posterior y dentro de N meses calendario del primero
--   madurez       = solo cuenta a la persona si primer_pedido + N meses <= última fecha de datos
-- Nota: no se ejecutó contra un SQL Server real; la lógica se validó en DuckDB (ver duckdb/).
-- ============================================================================

-- A) Venta neta mensual por canal con variación frente al mes anterior
;WITH mensual AS (
    SELECT DATEFROMPARTS(YEAR(fecha_pedido_dia), MONTH(fecha_pedido_dia), 1) AS mes,
           canal,
           SUM(valor_neto) AS venta_neta,
           COUNT(*)        AS pedidos
    FROM silver.pedidos
    WHERE estado = 'ENTREGADO'
    GROUP BY DATEFROMPARTS(YEAR(fecha_pedido_dia), MONTH(fecha_pedido_dia), 1), canal
), con_anterior AS (
    SELECT mes, canal, pedidos, venta_neta,
           LAG(venta_neta) OVER (PARTITION BY canal ORDER BY mes) AS venta_mes_anterior
    FROM mensual
)
SELECT mes, canal, pedidos, venta_neta, venta_mes_anterior,
       ROUND(100.0 * (venta_neta - venta_mes_anterior) / NULLIF(venta_mes_anterior, 0), 2) AS variacion_pct
FROM con_anterior
ORDER BY canal, mes;

-- B) Recompra por cohorte de registro a 1, 3 y 6 meses
;WITH personas AS (
    SELECT m.id_cliente_maestro AS persona, MIN(c.cohorte_registro) AS cohorte
    FROM silver.clientes c
    JOIN silver.v_cliente_maestro m ON m.id_cliente = c.id_cliente
    GROUP BY m.id_cliente_maestro
), entregados AS (
    SELECT m.id_cliente_maestro AS persona, p.fecha_pedido_dia AS fecha
    FROM silver.pedidos p
    JOIN silver.v_cliente_maestro m ON m.id_cliente = p.id_cliente
    WHERE p.estado = 'ENTREGADO'
), primer AS (
    SELECT persona, MIN(fecha) AS fecha_primero FROM entregados GROUP BY persona
), ventanas AS (
    SELECT pr.persona, pr.fecha_primero,
           MAX(CASE WHEN e.fecha > pr.fecha_primero AND e.fecha <= DATEADD(MONTH, 1, pr.fecha_primero) THEN 1 ELSE 0 END) AS r1,
           MAX(CASE WHEN e.fecha > pr.fecha_primero AND e.fecha <= DATEADD(MONTH, 3, pr.fecha_primero) THEN 1 ELSE 0 END) AS r3,
           MAX(CASE WHEN e.fecha > pr.fecha_primero AND e.fecha <= DATEADD(MONTH, 6, pr.fecha_primero) THEN 1 ELSE 0 END) AS r6
    FROM primer pr
    JOIN entregados e ON e.persona = pr.persona
    GROUP BY pr.persona, pr.fecha_primero
), limite AS (
    SELECT MAX(fecha_pedido_dia) AS ultima FROM silver.pedidos
)
SELECT p.cohorte,
       COUNT(*)         AS personas_cohorte,
       COUNT(v.persona) AS compradores,
       SUM(CASE WHEN DATEADD(MONTH, 1, v.fecha_primero) <= l.ultima THEN 1 ELSE 0 END)       AS base_1m,
       SUM(CASE WHEN DATEADD(MONTH, 1, v.fecha_primero) <= l.ultima THEN v.r1  ELSE 0 END)  AS recompra_1m,
       ROUND(100.0 * SUM(CASE WHEN DATEADD(MONTH, 1, v.fecha_primero) <= l.ultima THEN v.r1  ELSE 0 END)
             / NULLIF(SUM(CASE WHEN DATEADD(MONTH, 1, v.fecha_primero) <= l.ultima THEN 1 ELSE 0 END), 0), 2) AS tasa_1m,
       SUM(CASE WHEN DATEADD(MONTH, 3, v.fecha_primero) <= l.ultima THEN 1 ELSE 0 END)       AS base_3m,
       SUM(CASE WHEN DATEADD(MONTH, 3, v.fecha_primero) <= l.ultima THEN v.r3  ELSE 0 END)  AS recompra_3m,
       ROUND(100.0 * SUM(CASE WHEN DATEADD(MONTH, 3, v.fecha_primero) <= l.ultima THEN v.r3  ELSE 0 END)
             / NULLIF(SUM(CASE WHEN DATEADD(MONTH, 3, v.fecha_primero) <= l.ultima THEN 1 ELSE 0 END), 0), 2) AS tasa_3m,
       SUM(CASE WHEN DATEADD(MONTH, 6, v.fecha_primero) <= l.ultima THEN 1 ELSE 0 END)       AS base_6m,
       SUM(CASE WHEN DATEADD(MONTH, 6, v.fecha_primero) <= l.ultima THEN v.r6 ELSE 0 END)  AS recompra_6m,
       ROUND(100.0 * SUM(CASE WHEN DATEADD(MONTH, 6, v.fecha_primero) <= l.ultima THEN v.r6 ELSE 0 END)
             / NULLIF(SUM(CASE WHEN DATEADD(MONTH, 6, v.fecha_primero) <= l.ultima THEN 1 ELSE 0 END), 0), 2) AS tasa_6m
FROM personas p
LEFT JOIN ventanas v ON v.persona = p.persona
CROSS JOIN limite l
GROUP BY p.cohorte
ORDER BY p.cohorte;

"""Orquestación: clientes -> pedidos -> pagos -> venta neta diaria.

Orden importa: los pagos se validan contra los pedidos ya cargados, y los pedidos
contra los clientes. Toda la corrida es una sola transacción: si el proceso se cae a la
mitad no queda nada a medias (Silver sin Gold) y basta con volver a ejecutarlo. Lo único
que sobrevive a una caída es el Parquet de Bronze, que se vuelve a escribir igual.
"""
from __future__ import annotations

import uuid
from datetime import datetime, timedelta
from pathlib import Path

import pandas as pd

from . import bronze, rules, store, tickets
from .operations import bloqueo, actualizar_incidencias, validar_publicacion
from .star import refrescar_estrella
from .config import ESTADO_VENTA, Config


def ejecutar(raw_dir, db_path, cfg=None, bronze_dir=None, recalcular_gold=False,
             politica=None, reconciliar=False):
    """Escritor exclusivo local. Reconciliar relee pagos, nunca infiere borrados."""
    with bloqueo(db_path):
        return _ejecutar(raw_dir, db_path, cfg, bronze_dir, recalcular_gold, politica, reconciliar)


def _ejecutar(raw_dir: str | Path, db_path: str | Path, cfg: Config | None = None,
             bronze_dir: str | Path | None = None, recalcular_gold: bool = False, politica=None, reconciliar: bool = False) -> dict:
    """recalcular_gold: recalcula todos los días de Gold, no solo los tocados. Hace falta
    cuando cambia la definición de venta (por ejemplo, atribuir a fecha_entrega) sin que
    cambien los datos."""
    cfg = cfg or Config()
    raw_dir = Path(raw_dir)
    bronze_dir = Path(bronze_dir) if bronze_dir else Path(db_path).parent / "bronze"
    run_id = str(uuid.uuid4())
    inicio = datetime.now()
    con = store.conectar(str(db_path))
    resumen: dict = {"run_id": run_id, "bronze": {}}

    def fuente(tabla: str) -> pd.DataFrame:
        # CSV -> Bronze (Parquet crudo, inmutable) -> Silver. Silver nunca lee el CSV directo.
        df, info = bronze.aterrizar(con, raw_dir, bronze_dir, tabla, run_id)
        resumen["bronze"][tabla] = info
        return df

    con.begin()
    try:
        # ---- clientes (dimensión pequeña: se lee completa; el hash evita reescrituras)
        cli_raw = fuente("clientes")
        cli_ok, cli_q = rules.validar_clientes(cli_raw, cfg)
        resumen["clientes"], _ = store.upsert(con, "silver_clientes", cli_ok)
        resumen["clientes"]["leidos"] = len(cli_raw)
        resumen["clientes"]["cuarentena_nueva"] = store.registrar_cuarentena(con, cli_q, run_id)
        store.retirar(con, "silver_clientes", _rechazados(cli_q, cli_ok, "id_cliente"))
        actualizar_incidencias(con, 'clientes', cli_raw, cli_q, run_id)
        resumen['clientes']['rechazados_lote'] = len(cli_raw) - len(cli_ok)
        clientes_ids = set(con.execute("SELECT id_cliente FROM silver_clientes").df()["id_cliente"])

        # ---- pedidos: la fuente no trae fecha de carga, así que el incremental se hace
        # por contenido: solo se escribe lo nuevo o lo que cambió de hash.
        ped_raw = fuente("pedidos")
        ped_ok, ped_q = rules.validar_pedidos(
            ped_raw, cfg, clientes_ids, store.negocio_existente(con, "silver_pedidos"))
        dias_antes = con.execute("SELECT id_pedido, fecha_pedido_dia FROM silver_pedidos").df()
        resumen["pedidos"], ped_escritos = store.upsert(con, "silver_pedidos", ped_ok)
        resumen["pedidos"]["leidos"] = len(ped_raw)
        resumen["pedidos"]["cuarentena_nueva"] = store.registrar_cuarentena(con, ped_q, run_id)
        # Un pedido que ya estaba en Silver y hoy es inválido (dato corregido en la fuente,
        # regla nueva, rango más estrecho) sale de Silver y deja de sumar venta.
        ped_retirados = store.retirar(con, "silver_pedidos", _rechazados(ped_q, ped_ok, "id_pedido"))
        resumen["pedidos"]["retirados"] = len(ped_retirados)

        actualizar_incidencias(con, 'pedidos', ped_raw, ped_q, run_id)
        resumen['pedidos']['rechazados_lote'] = len(ped_raw) - len(ped_ok)
        con.execute("UPDATE silver_pedidos SET flag_cliente_desconocido = NOT EXISTS "
                    "(SELECT 1 FROM silver_clientes c WHERE c.id_cliente=silver_pedidos.id_cliente)")

        # Días de venta a recalcular: el día nuevo de cada fila escrita y, si la fila ya
        # existía, también su día anterior (por si el pedido cambió de fecha), más los días
        # de los pedidos retirados.
        fechas_afectadas = _fechas_afectadas(ped_escritos, dias_antes, ped_retirados)

        # ---- pagos: incremental por watermark de fecha_carga, con ventana de reproceso
        pag_raw = fuente("pagos")
        wm = store.leer_watermark(con, "pagos")
        if wm is not None and not reconciliar:
            desde = pd.Timestamp(wm) - timedelta(days=cfg.lookback_dias)
            carga = rules._a_fecha(pag_raw["fecha_carga"])
            # Los pagos que esperan a su pedido se releen siempre: si el pedido llega después
            # de que el pago salió de la ventana, de otro modo quedarían huérfanos para siempre.
            pendiente = pag_raw["id_pago"].astype(str).str.strip().isin(store.pagos_pendientes(con))
            pag_raw = pag_raw[(carga >= desde) | carga.isna() | pendiente].reset_index(drop=True)
        pedidos_dia = con.execute(
            "SELECT id_pedido, fecha_pedido_dia FROM silver_pedidos").df().set_index("id_pedido")["fecha_pedido_dia"]
        pedidos_dia = pd.to_datetime(pedidos_dia)
        pag_ok, pag_q = rules.validar_pagos(
            pag_raw, cfg, pedidos_dia, store.ids_en_cuarentena(con, "pedidos"),
            store.negocio_existente(con, "silver_pagos"))
        resumen["pagos"], _ = store.upsert(con, "silver_pagos", pag_ok)
        resumen["pagos"]["leidos"] = len(pag_raw)
        resumen["pagos"]["cuarentena_nueva"] = store.registrar_cuarentena(con, pag_q, run_id)
        resumen["pagos"]["retirados"] = len(store.retirar(con, "silver_pagos", _rechazados(pag_q, pag_ok, "id_pago")))
        actualizar_incidencias(con, 'pagos', pag_raw, pag_q, run_id)
        resumen['pagos']['rechazados_lote'] = len(pag_raw) - len(pag_ok)
        if len(pag_raw):
            maximo = rules._a_fecha(pag_raw["fecha_carga"]).max()
            if pd.notna(maximo):
                previo = pd.Timestamp(wm) if wm is not None else maximo
                store.escribir_watermark(con, "pagos", max(maximo, previo).date(), run_id)

        # Soporte es opcional y no altera el cálculo de venta.
        con.execute(tickets.ESQUEMA)
        if (raw_dir / 'tickets_soporte.csv').exists():
            ticket_raw = fuente('tickets_soporte')
            ticket_ok, ticket_q = tickets.validar_tickets(ticket_raw)
            resumen['tickets_soporte'], _ = store.upsert(con, 'silver_tickets', ticket_ok)
            store.retirar(con, 'silver_tickets', _rechazados(ticket_q, ticket_ok, 'id_ticket'))
            resumen['tickets_soporte'].update(leidos=len(ticket_raw),
                rechazados_lote=len(ticket_raw)-len(ticket_ok),
                cuarentena_nueva=store.registrar_cuarentena(con, ticket_q, run_id))
            actualizar_incidencias(con, 'tickets_soporte', ticket_raw, ticket_q, run_id)

        # ---- gold: venta neta diaria solo para los días tocados (o todos, si se pide)
        if recalcular_gold:
            fechas_afectadas = store.dias_con_venta(con)
        resumen["dias_recalculados"] = store.refrescar_venta_neta(con, fechas_afectadas, ESTADO_VENTA)

        resumen['publicacion'] = validar_publicacion(con, resumen, politica)
        resumen['estrella'] = refrescar_estrella(con, cfg)
        con.execute("CREATE TABLE IF NOT EXISTS publicacion_actual "
                    "(id INTEGER PRIMARY KEY, run_id VARCHAR, validado_en TIMESTAMP, venta_neta BIGINT)")
        con.execute("INSERT OR REPLACE INTO publicacion_actual VALUES (1,?,current_timestamp,?)",
                    [run_id, resumen['publicacion']['venta_neta']])
        _log(con, run_id, inicio, "OK", resumen)
        con.commit()
    except Exception:
        con.rollback()
        _log(con, run_id, inicio, "FALLIDO", resumen)
        raise
    finally:
        con.close()
    return resumen


def _rechazados(cuarentena: pd.DataFrame, validos: pd.DataFrame, clave: str) -> set[str]:
    """Ids que en esta corrida fueron a cuarentena y no quedaron entre los válidos."""
    ids = {i.strip() for i in cuarentena["id_registro"] if i.strip()}
    return ids - set(validos[clave])


def _fechas_afectadas(escritos: pd.DataFrame, dias_antes: pd.DataFrame,
                      retirados: pd.DataFrame | None = None) -> list:
    dias = set()
    if not escritos.empty:
        dias |= set(pd.to_datetime(escritos["fecha_pedido_dia"]).dt.date)
        previos = dias_antes[dias_antes["id_pedido"].isin(escritos["id_pedido"])]
        dias |= set(pd.to_datetime(previos["fecha_pedido_dia"]).dt.date)

/* ---------------------------------------------------------------------------
   PROBLEMAS DE LA CONSULTA ORIGINAL
   1. Subconsulta correlacionada para venta_mes: se ejecuta una vez por cliente.
   2. YEAR(p.fecha_pedido) / MONTH(p.fecha_pedido): función sobre la columna; el índice por
      fecha no se puede usar (predicado no SARGable) y se lee la tabla completa.
   3. CAST(... AS VARCHAR(20)) en ambos lados del cruce con tickets_soporte: conversión
      sobre las dos columnas, ninguna búsqueda por índice. Ambos id_cliente ya son VARCHAR(9)
      (ver DDL): el CAST no aporta nada y es lo que más pesa.
   4. IN (SELECT DISTINCT ...): buena práctica, no causa de lentitud. El optimizador suele
      convertirlo en semiunión y descartar el DISTINCT; EXISTS expresa mejor la intención.
   5. TOP 1 ... ORDER BY fecha_creacion DESC sin índice (id_cliente, fecha_creacion DESC):
      ordena los tickets del cliente en cada ejecución.
   6. GETDATE(): SQL Server lo evalúa una vez por consulta, no por fila; se pasa a una
      variable para que el rango del mes sea explícito y se pueda probar con otra fecha.
   7. Observación de negocio: el original suma TODOS los estados del mes (también devueltos y
      cancelados). La reescritura conserva ese resultado para ser equivalente; si la intención
      es venta neta, añade AND p.estado = 'ENTREGADO' (se señala abajo).
--------------------------------------------------------------------------- */

-- CONSULTA REESCRITA (mismo resultado que la original)
DECLARE @ini DATE = DATEFROMPARTS(YEAR(GETDATE()), MONTH(GETDATE()), 1);
DECLARE @fin DATE = DATEADD(MONTH, 1, @ini);

WITH venta AS (
    SELECT p.id_cliente,
           SUM(p.valor_bruto - p.descuento) AS venta_mes
    FROM dbo.pedidos AS p
    WHERE p.fecha_pedido >= @ini
      AND p.fecha_pedido <  @fin          -- rango SARGable: usa el índice por fecha
      -- AND p.estado = 'ENTREGADO'       -- descomentar si se quiere venta neta (ver punto 7)
    GROUP BY p.id_cliente
)
SELECT c.id_cliente,
       v.venta_mes,
       u.canal AS ultimo_canal_soporte
FROM dbo.clientes AS c
LEFT JOIN venta AS v ON v.id_cliente = c.id_cliente
OUTER APPLY (
    SELECT TOP (1) t.canal
    FROM dbo.tickets_soporte AS t
    WHERE t.id_cliente = c.id_cliente     -- mismo tipo a ambos lados: búsqueda por índice
    ORDER BY t.fecha_creacion DESC
) AS u
WHERE EXISTS (
    SELECT 1 FROM dbo.pedidos AS d
    WHERE d.id_cliente = c.id_cliente AND d.estado = 'DEVUELTO'
)
ORDER BY v.venta_mes DESC;
GO

"""Reporte diario de venta neta por canal (versión corregida del PR del analista junior).

Venta neta = valor_bruto - descuento de pedidos ENTREGADO, por fecha_pedido, en hora de Colombia.
Lee la tabla certificada gold.fact_venta_neta_diaria (la misma que alimenta Power BI), no la
tabla cruda dbo.pedidos: ahí los canales llegan con variantes ('app ', 'PAGINA WEB') y hay
pedidos duplicados, y el reporte daría una cifra distinta a la oficial.
"""
from __future__ import annotations

import logging
import os
import sys
from contextlib import closing
from dataclasses import dataclass
from datetime import date, datetime, timedelta
from zoneinfo import ZoneInfo

log = logging.getLogger(__name__)

CANALES = ("WEB", "APP", "TIENDA")
ZONA = ZoneInfo("America/Bogota")

# Todo el cálculo ocurre en el motor de base de datos: una consulta, parámetros enlazados
# (sin SQL armado con f-strings) y solo las columnas necesarias (sin SELECT *).
# Gold ya trae solo pedidos ENTREGADO, el canal normalizado y sin duplicados.
CONSULTA = """
SELECT c.canal,
       SUM(v.venta_neta)  AS total,
       SUM(v.num_pedidos) AS pedidos
FROM gold.fact_venta_neta_diaria v
JOIN gold.dim_canal c ON c.canal_key = v.canal_key
WHERE v.fecha_key = ?
GROUP BY c.canal
"""


@dataclass(frozen=True)
class VentaCanal:
    canal: str
    total: int
    pedidos: int

    @property
    def ticket(self) -> int | None:
        """Ticket promedio; None (no 0) si no hubo pedidos, para no mostrar un dato falso."""
        return round(self.total / self.pedidos) if self.pedidos else None


def ventas_por_canal(conn, dia: date) -> dict[str, VentaCanal]:
    """Venta neta y pedidos entregados de `dia`. Todos los canales aparecen, aunque no tengan ventas."""
    cur = conn.cursor()
    cur.execute(CONSULTA, (int(dia.strftime("%Y%m%d")),))  # fecha_key = yyyymmdd
    encontrados = {fila[0]: VentaCanal(fila[0], round(fila[1] or 0), int(fila[2])) for fila in cur.fetchall()}
    return {c: encontrados.get(c, VentaCanal(c, 0, 0)) for c in CANALES}
